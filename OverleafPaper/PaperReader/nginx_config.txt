
    # ============================================
    # PaperReader - Academic Paper Viewer (port 22222 tunneled)

    location /paperreader/ {
        alias /mnt/data/Yanlai/paperreader/;
        index index.html;
        try_files $uri $uri/ /paperreader/index.html;
    }

    location /paperreader/api/ {
        rewrite ^/paperreader/api/(.*) /api/$1 break;
        proxy_pass http://127.0.0.1:22222;
        proxy_http_version 1.1;

        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;

        proxy_connect_timeout 60s;
        proxy_send_timeout 300s;
        proxy_read_timeout 300s;

        client_max_body_size 100M;

        proxy_buffering on;
        proxy_buffer_size 128k;
        proxy_buffers 8 128k;
        proxy_busy_buffers_size 256k;
    }

    location /paperreader/ws/ {
        rewrite ^/paperreader/ws/(.*) /ws/$1 break;
        proxy_pass http://127.0.0.1:22222;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_set_header Host $host;
        proxy_read_timeout 86400; # Keep connection alive for 24h
    }

    location /paperreader/health {
        proxy_pass http://127.0.0.1:22222/health;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
    }
